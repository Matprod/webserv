cat > /tmp/nginx-webserv.conf <<'NGINX'
worker_processes auto;
events { worker_connections 1024; }
http {
	server {
		listen 8080;
		root /srv/www;
		autoindex on;

		location /readOnly/ {
			limit_except GET { deny all; }
		}
	}
}
NGINX

//42 COMMAND
docker run --rm -p 8080:8080 \
  -v /tmp/nginx-webserv.conf:/etc/nginx/nginx.conf:ro \
  -v /home/adebert/Project/webserv/srv/www:/srv/www:ro \
  nginx:alpine
